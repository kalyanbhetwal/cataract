# Windows PowerShell commands. Run one command at a time from this project directory.
# Replace the HEDS path, camera serial, and wavelength with the values for your setup.
# "horizontal" means horizontal fringe lines; the phase ramp varies top-to-bottom along y.
# If you instead need horizontal beam steering along x, use vertical fringe lines.
# Current workflow: capture the fringe alone first, then add random phase to the same fringe.
# Save the full camera field so zero and first orders are included when both are on the sensor.
# --num-patterns counts random-plus-fringe frames; --capture-fringe-first adds one reference frame.

# Install Python dependencies.
python -m pip install -r requirements.txt

# Simulate the full sequence: one horizontal-fringe reference plus 3 random-plus-fringe frames (4 saved frames total).
python acquire_slm_camera.py --dry-run --capture-fringe-first --num-patterns 3 --slm-width 1920 --slm-height 1200 --fringe-orientation horizontal --settle-ms 300 --discard-frames 2 --no-neuws-processing --output-format npy --save-full-patterns

# Run the automated software tests without hardware.
python -m unittest -v test_acquire_slm_camera.py

# Capture one real camera frame without opening the SLM.
python camera_smoke_test.py --camera-serial YOUR_CAMERA_SERIAL --exposure-ms 100 --gain-db 0 --pixel-format Mono16

# Tell the current PowerShell session where the HOLOEYE SDK examples directory is located.
$env:HEDS_EXAMPLES_DIR = "C:\Program Files\HOLOEYE Photonics\SLM Display SDK (Python) v4.2.1\examples"

# Optional fringe-only check: capture both orders in one full frame with no random phase. Open data\find_steered_order\previews\frame_0001.png.
python acquire_slm_camera.py --num-patterns 1 --coefficient-std-rad 0 --slm-width 1920 --slm-height 1200 --fringe-orientation horizontal --settle-ms 300 --discard-frames 2 --slm-preselect "index:0" --camera-serial YOUR_CAMERA_SERIAL --wavelength-nm 532 --exposure-ms 100 --gain-db 0 --pixel-format Mono16 --no-neuws-processing --save-full-patterns --output-format npy --output-dir data\find_steered_order

# Capture both orders: one pure horizontal-fringe reference, then 3 random-plus-fringe measurements (4 saved frames total). Each SLM update settles for 300 ms before 2 buffered frames are discarded.
python acquire_slm_camera.py --capture-fringe-first --num-patterns 3 --slm-width 1920 --slm-height 1200 --fringe-orientation horizontal --settle-ms 300 --discard-frames 2 --no-neuws-processing --slm-preselect "index:0" --camera-serial YOUR_CAMERA_SERIAL --wavelength-nm 532 --exposure-ms 100 --gain-db 0 --pixel-format Mono16 --save-full-patterns --output-dir data\hardware_test_001

# Capture both orders for a full run: one fringe reference plus 100 random-plus-fringe measurements (101 saved frames total). Use a new output directory each time.
python acquire_slm_camera.py --capture-fringe-first --num-patterns 100 --slm-width 1920 --slm-height 1200 --fringe-orientation horizontal --settle-ms 300 --discard-frames 2 --no-neuws-processing --slm-preselect "index:0" --camera-serial YOUR_CAMERA_SERIAL --wavelength-nm 532 --exposure-ms 100 --gain-db 0 --pixel-format Mono16 --save-full-patterns --output-dir data\measurement_001
